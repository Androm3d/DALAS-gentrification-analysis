def merge_csv_per_year(name, yearStart, yearEnd, input_dir, output_path=None, verbose=True):
    """
    Merge yearly files named {year}_{name}.csv for years 2015-2025
    into a single CSV named yearStart-yearEnd_{name}.csv by default.

    Parameters:
    - input_dir: path to the folder containing the yearly CSVs (string or Path)
    - output_path: optional path (string or Path) for the merged CSV file
    - verbose: print progress/messages if True

    Returns:
    - pandas.DataFrame with the concatenated data
    """
    input_dir = Path(input_dir)
    years = range(yearStart, yearEnd + 1)
    dfs = []
    for y in years:
        fp = input_dir / f"{y}_{name}.csv"
        if fp.exists():
            try:
                df = pd.read_csv(fp)
                df = df.iloc[:, 1:]     # Drop first column since it's a date 
                df['year'] = y
                dfs.append(df)
                if verbose:
                    print(f"Loaded: {fp.name} ({len(df)} rows)")
            except Exception as e:
                if verbose:
                    print(f"Failed to read {fp}: {e}")
        else:
            if verbose:
                print(f"Missing: {fp.name}")
    if not dfs:
        raise FileNotFoundError("No input files found for 2015-2025 in " + str(input_dir))
    out_df = pd.concat(dfs, ignore_index=True, sort=False)
    out_path = Path(output_path) if output_path else input_dir / f"{yearStart}-{yearEnd}_{name}.csv"
    out_df.to_csv(out_path, index=False)
    if verbose:
        print(f"Saved merged CSV to: {out_path} ({len(out_df)} total rows)")
    return out_df

def smart_fill_prices_adaptive(reg_df, mkt_df, name_to_id_map):
    print("--- Running Adaptive Regression Fill ---")
    
    # 1. Helper to clean and melt (Same as before)
    def clean(df, source):
        df = df.copy()
        if 'Tipus de territori' in df.columns: df = df[df['Tipus de territori'] == 'Barri']
        if 'Territori' in df.columns: df.rename(columns={'Territori': 'Nom_barri'}, inplace=True)
        
        manual_corrections = {
             'el Poble Sec - AEI Parc Montjuïc': 'el Poble Sec',
             'Sant Andreu (Barri)': 'Sant Andreu',
             'Sants Badal': 'Sants-Badal',
             'Sant Gervasi Galvany': 'Sant Gervasi - Galvany'
        }
        df['Nom_barri'] = df['Nom_barri'].replace(manual_corrections)
        if 'Codi_barri' not in df.columns: df['Codi_barri'] = df['Nom_barri'].map(name_to_id_map)
        
        if source == 'reg':
            cols = [c for c in df.columns if 'tr ' in c]
            long = df.melt(id_vars=['Codi_barri'], value_vars=cols, value_name='Price')
            long['year'] = long['variable'].str.extract(r'(\d{4})').astype(float)
        else:
            cols = [c for c in df.columns if str(c).isdigit()]
            long = df.melt(id_vars=['Codi_barri'], value_vars=cols, var_name='year', value_name='Price')
            long['year'] = long['year'].astype(float)
            
        long['Price'] = pd.to_numeric(long['Price'], errors='coerce')
        return long.groupby(['Codi_barri', 'year'])['Price'].mean().reset_index()

    df_reg = clean(reg_df, 'reg')
    df_mkt = clean(mkt_df, 'mkt')
    
    # 2. Train the Adapter (Linear Regression)
    # We only use years where we have BOTH data to learn the relationship
    common = pd.merge(df_reg, df_mkt, on=['Codi_barri', 'year'], suffixes=('_reg', '_mkt')).dropna()
    
    X = common[['Price_mkt']] # Feature: Market Price
    y = common['Price_reg']   # Target: Real Price
    
    model = LinearRegression()
    model.fit(X, y)
    
    r2 = model.score(X, y)
    print(f"Adaptive Model trained on {len(common)} overlaps.")
    print(f"Relationship R2: {r2:.4f}")
    print(f"Formula: Real = {model.intercept_:.2f} + {model.coef_[0]:.4f} * Market")
    
    # 3. Apply Prediction to Missing Data
    # Merge everything
    final = pd.merge(df_reg, df_mkt, on=['Codi_barri', 'year'], suffixes=('_reg', '_mkt'), how='outer')
    
    # Identify rows needing fill (Reg is missing, Mkt exists)
    needs_fill = final['Price_reg'].isna() & final['Price_mkt'].notna()
    
    # Predict ONLY for those rows
    if needs_fill.sum() > 0:
        # We must reshape for sklearn
        X_missing = final.loc[needs_fill, ['Price_mkt']]
        predicted_real = model.predict(X_missing)
        
        # Fill
        final.loc[needs_fill, 'Price_reg'] = predicted_real
    
    # 4. Final Cleanup
    final['median_price_per_m2'] = final['Price_reg']
    final['is_price_imputed'] = needs_fill.astype(int)
    
    print(f"Filled {needs_fill.sum()} missing years using Adaptive Regression.")
    
    return final[['Codi_barri', 'year', 'median_price_per_m2', 'is_price_imputed']]

def diagnose_price_mismatch(reg_df, mkt_df, name_to_id_map):
    print("--- Starting Diagnosis ---")
    
    # --- HELPER: Smart Preprocessing ---
    def prepare_df(df, source_type):
        df = df.copy()
        
        # 1. Handle "Tipus de territori" (If it exists, filter it. If not, assume already filtered)
        if 'Tipus de territori' in df.columns:
            df = df[df['Tipus de territori'] == 'Barri'].copy()
        
        # 2. Handle Name Column (Raw 'Territori' vs Clean 'Nom_barri')
        if 'Territori' in df.columns:
            df.rename(columns={'Territori': 'Nom_barri'}, inplace=True)
            
        # 3. Clean Names (Apply corrections if needed)
        manual_corrections = {
             'el Poble Sec - AEI Parc Montjuïc': 'el Poble Sec',
             'Sant Andreu (Barri)': 'Sant Andreu',
             'Sants Badal': 'Sants-Badal',
             'Sant Gervasi Galvany': 'Sant Gervasi - Galvany'
        }
        df['Nom_barri'] = df['Nom_barri'].replace(manual_corrections)
        
        # 4. Ensure ID exists
        if 'Codi_barri' not in df.columns:
             df['Codi_barri'] = df['Nom_barri'].map(name_to_id_map)
        
        # 5. MELT (Wide to Long)
        if source_type == 'registered':
            # Identify quarter columns (e.g., '1r tr 2014')
            val_cols = [c for c in df.columns if 'tr ' in c]
            df_long = df.melt(id_vars=['Codi_barri'], value_vars=val_cols, value_name='Price')
            # Extract Year from string "1r tr 2014"
            df_long['year'] = df_long['variable'].str.extract(r'(\d{4})').astype(float)
        else:
            # Identify year columns (digits only, like '2015')
            val_cols = [c for c in df.columns if str(c).isdigit()]
            df_long = df.melt(id_vars=['Codi_barri'], value_vars=val_cols, var_name='year', value_name='Price')
            df_long['year'] = df_long['year'].astype(float)
            
        # 6. Clean Numerics (Crucial fix for "String Sum" error)
        df_long['Price'] = pd.to_numeric(df_long['Price'], errors='coerce')
        
        # 7. Aggregate to Yearly Mean
        return df_long.groupby(['Codi_barri', 'year'])['Price'].mean().reset_index()

    # --- EXECUTE ---
    reg_yearly = prepare_df(reg_df, 'registered')
    mkt_yearly = prepare_df(mkt_df, 'market')
    
    # --- MERGE & ANALYZE ---
    merged = pd.merge(reg_yearly, mkt_yearly, on=['Codi_barri', 'year'], suffixes=('_Reg', '_Mkt'), how='inner')
    
    # Drop NaNs
    valid_data = merged.dropna(subset=['Price_Reg', 'Price_Mkt'])
    
    if len(valid_data) > 0:
        corr, _ = pearsonr(valid_data['Price_Reg'], valid_data['Price_Mkt'])
        valid_data['Gap_Pct'] = (valid_data['Price_Mkt'] - valid_data['Price_Reg']) / valid_data['Price_Reg'] * 100
        
        print(f"\n--- REPORT ---")
        print(f"Data Points: {len(valid_data)}")
        print(f"Correlation: {corr:.4f}")
        print(f"Avg Gap (Market > Real): {valid_data['Gap_Pct'].mean():.2f}%")
        print(f"Gap Volatility (Std Dev): {valid_data['Gap_Pct'].std():.2f}%")
        
        # Visualization
        plt.figure(figsize=(10, 5))
        
        # Plot 1: Scatter
        plt.subplot(1, 2, 1)
        sns.scatterplot(data=valid_data, x='Price_Reg', y='Price_Mkt', alpha=0.5)
        max_val = max(valid_data['Price_Reg'].max(), valid_data['Price_Mkt'].max())
        plt.plot([0, max_val], [0, max_val], 'r--', label="Identity")

        # Plot 2: Trajectories

            plt.plot(subset['year'], subset['Price_Reg'], label=f"Reg {bid}")
            plt.plot(subset['year'], subset['Price_Mkt'], linestyle='--', label=f"Mkt {bid}")
        plt.title("Trends (Solid=Real, Dashed=Market)")
        plt.legend()
        plt.tight_layout()
        plt.show()
    else:
        print("⚠️ No overlapping data found (Check if years or IDs match).")
    
    return merged
